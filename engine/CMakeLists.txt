# Native Windows build of the Vulkan engine and its replay runners (Linux uses build.sh).
# The engine is a static library here: its C++ classes are not annotated for DLL export.
cmake_minimum_required(VERSION 3.20)
project(d4r_engine LANGUAGES CXX)
find_package(Vulkan REQUIRED)

add_library(d4r_engine STATIC runtime.cpp k_frame.cpp game.cpp game_m.cpp game_rr.cpp game_e.cpp)
target_compile_features(d4r_engine PUBLIC cxx_std_17)
target_include_directories(d4r_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(d4r_engine PUBLIC Vulkan::Vulkan)
if(MSVC)
    target_compile_definitions(d4r_engine PUBLIC NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(d4r_engine PUBLIC /W3 /permissive- /utf-8 /bigobj)
else()
    target_compile_options(d4r_engine PUBLIC -Wall -Wextra -Wno-missing-field-initializers)
endif()

# The native HIP network (hip_net.h) for d4r-k and d4r-m's `hip` argument, as engine/build.sh adds it on Linux:
# found through D4R_ROCM_DIR, HIP_PATH or ROCM_PATH (on Windows, AMD's HIP SDK, e.g. C:/ROCm/7.1).
set(D4R_ROCM_DIR "$ENV{D4R_ROCM_DIR}" CACHE PATH "ROCm / HIP SDK root for the native HIP network (optional)")
if(NOT D4R_ROCM_DIR)
    foreach(CANDIDATE "$ENV{HIP_PATH}" "$ENV{ROCM_PATH}" /opt/rocm)
        if(CANDIDATE AND EXISTS "${CANDIDATE}/include/hip/hip_runtime_api.h")
            set(D4R_ROCM_DIR "${CANDIDATE}")
            break()
        endif()
    endforeach()
endif()
if(NOT D4R_ROCM_DIR AND WIN32)
    file(GLOB D4R_ROCM_CANDIDATES "C:/ROCm/*" "C:/Program Files/AMD/ROCm/*")
    foreach(CANDIDATE IN LISTS D4R_ROCM_CANDIDATES)
        if(EXISTS "${CANDIDATE}/include/hip/hip_runtime_api.h")
            set(D4R_ROCM_DIR "${CANDIDATE}") # the newest, by name
        endif()
    endforeach()
endif()
find_library(D4R_AMDHIP64 NAMES amdhip64 HINTS "${D4R_ROCM_DIR}/lib" NO_DEFAULT_PATH)
if(D4R_ROCM_DIR AND D4R_AMDHIP64 AND EXISTS "${D4R_ROCM_DIR}/include/hip/hip_runtime_api.h")
    add_library(d4r_hip_net STATIC hip_net.cpp)
    target_compile_features(d4r_hip_net PUBLIC cxx_std_17)
    target_include_directories(d4r_hip_net PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${D4R_ROCM_DIR}/include")
    target_compile_definitions(d4r_hip_net PUBLIC D4R_ENGINE_HIP __HIP_PLATFORM_AMD__)
    target_link_libraries(d4r_hip_net PUBLIC "${D4R_AMDHIP64}")
    message(STATUS "d4r-k/d4r-m: native HIP network from ${D4R_ROCM_DIR}")
else()
    message(STATUS "d4r-k/d4r-m: no HIP SDK found, Vulkan network only (set D4R_ROCM_DIR)")
endif()

# vkhip (vkhip/vkhip_net.h) for d4r-m's `vkhip` argument, as engine/build.sh adds it on Linux: the HIP code objects
# as Vulkan pipelines (through VK_KHR_pipeline_binary on AMD's Windows driver). Its donor shader needs glslangValidator.
find_program(D4R_GLSLANG glslangValidator HINTS "$ENV{VULKAN_SDK}/Bin" "$ENV{VULKAN_SDK}/bin")
find_package(Python3 COMPONENTS Interpreter)
if(D4R_GLSLANG AND Python3_Interpreter_FOUND)
    set(D4R_VKHIP_DONOR "${CMAKE_CURRENT_BINARY_DIR}/vkhip/donor_spv.h")
    add_custom_command(OUTPUT "${D4R_VKHIP_DONOR}"
        COMMAND ${CMAKE_COMMAND} -E env "D4R_GLSLANG=${D4R_GLSLANG}" "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/build_vkhip_donor.py" "${CMAKE_CURRENT_BINARY_DIR}/vkhip"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/vkhip/donor.comp" "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/build_vkhip_donor.py"
        VERBATIM)
    add_library(d4r_vkhip STATIC vkhip/vkhip.cpp vkhip/vkhip_net.cpp "${D4R_VKHIP_DONOR}")
    target_compile_features(d4r_vkhip PUBLIC cxx_std_20)
    target_include_directories(d4r_vkhip PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/vkhip")
    target_compile_definitions(d4r_vkhip PUBLIC D4R_ENGINE_VKHIP)
    target_link_libraries(d4r_vkhip PUBLIC d4r_engine)
    message(STATUS "d4r-m: vkhip network (${D4R_GLSLANG})")
else()
    message(STATUS "d4r-m: no glslangValidator or Python, built without vkhip")
endif()

foreach(RUNNER IN ITEMS k m rr e)
    if(RUNNER STREQUAL "k")
        set(SOURCE replay.cpp)
    else()
        set(SOURCE replay_${RUNNER}.cpp)
    endif()
    add_executable(d4r-${RUNNER} ${SOURCE})
    target_link_libraries(d4r-${RUNNER} PRIVATE d4r_engine)
    if(TARGET d4r_hip_net AND (RUNNER STREQUAL "k" OR RUNNER STREQUAL "m"))
        target_link_libraries(d4r-${RUNNER} PRIVATE d4r_hip_net)
    endif()
    if(TARGET d4r_vkhip AND (RUNNER STREQUAL "m" OR RUNNER STREQUAL "rr"))
        target_link_libraries(d4r-${RUNNER} PRIVATE d4r_vkhip)
    endif()
endforeach()

add_executable(evk evk.cpp)
target_link_libraries(evk PRIVATE d4r_engine)

# The D3D12 backend (d3d12/): the same K frame (k_frame.cpp) and shaders (compiled to DXIL by hlsl_build.py).
if(WIN32)
    add_library(d4r_engine_d3d12 STATIC d3d12/runtime_d3d12.cpp d3d12/game_d3d12.cpp d3d12/runtime_m_d3d12.cpp
                d3d12/game_m_d3d12.cpp d3d12/runtime_rr_d3d12.cpp d3d12/game_rr_d3d12.cpp k_frame.cpp)
    target_compile_features(d4r_engine_d3d12 PUBLIC cxx_std_17)
    target_include_directories(d4r_engine_d3d12 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/d3d12")
    target_link_libraries(d4r_engine_d3d12 PUBLIC d3d12 dxgi)
    # d3dhip: the native HIP network (hip_net.h) between the D3D12 engine's stages, with the token/head buffers
    # shared as opaque NT handles (d3d12/hip_net_d3d12.h). The PCI identity of HIP device 0 comes from SetupAPI.
    if(TARGET d4r_hip_net)
        target_sources(d4r_engine_d3d12 PRIVATE d3d12/hip_net_d3d12.cpp d3d12/hip_net_rr.cpp)
        target_link_libraries(d4r_engine_d3d12 PUBLIC d4r_hip_net setupapi)
        target_compile_definitions(d4r_engine_d3d12 PUBLIC D4R_ENGINE_D3DHIP)
        message(STATUS "d3d12 engines: native HIP network (d3dhip) available")
        # The NGX shim's bridge (tools/d4r_nvngx_shim.cpp loads it on first use; the shim must not import amdhip64):
        # d4r_hip_net.dll with d4rHipNet* over hip_net_d3d12.h.
        add_library(d4r_hip_net_dll SHARED d3d12/hip_net_bridge.cpp d3d12/hip_net_d3d12.cpp)
        target_compile_features(d4r_hip_net_dll PRIVATE cxx_std_17)
        target_link_libraries(d4r_hip_net_dll PRIVATE d4r_hip_net d3d12 dxgi setupapi uuid)
        set_target_properties(d4r_hip_net_dll PROPERTIES OUTPUT_NAME d4r_hip_net ARCHIVE_OUTPUT_NAME d4r_hip_net_bridge)
    endif()
    if(MSVC)
        target_compile_definitions(d4r_engine_d3d12 PUBLIC NOMINMAX _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN)
        target_compile_options(d4r_engine_d3d12 PUBLIC /W3 /permissive- /utf-8)
    endif()
    add_executable(d4r-k-d3d12 d3d12/replay_d3d12.cpp)
    target_link_libraries(d4r-k-d3d12 PRIVATE d4r_engine_d3d12)
    # preset M: a standalone replay of the model's DXIL (compile_m.py --dxil)
    add_executable(d4r-m-d3d12 d3d12/replay_m_d3d12.cpp)
    target_link_libraries(d4r-m-d3d12 PRIVATE d4r_engine_d3d12)
    add_executable(d4r-rr-d3d12 d3d12/replay_rr_d3d12.cpp)
    target_link_libraries(d4r-rr-d3d12 PRIVATE d4r_engine_d3d12)
    add_executable(d4r-rr-zluda d3d12/replay_rr_zluda.cpp ../tools/d4r_ngx_param_msvc.cpp)
    target_compile_features(d4r-rr-zluda PRIVATE cxx_std_17)
    target_compile_definitions(d4r-rr-zluda PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    # preset M/L on RDNA 3 (no FP8 WMMA): FP16 WMMA with software FP8 storage, runtime_m_rdna3.inc
    add_executable(d4r-m-rdna3 d3d12/replay_m_rdna3.cpp)
    target_link_libraries(d4r-m-rdna3 PRIVATE d4r_engine_d3d12)
endif()
